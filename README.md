Example Queries
Top 5 Restaurants of Every City
SELECT name, city, (rating_count * cost) AS revenue 
FROM (
    SELECT name, city, rating_count, cost,
           ROW_NUMBER() OVER(PARTITION BY city ORDER BY rating_count * cost DESC) AS ranks
    FROM restaurants
) t
WHERE t.ranks < 6
ORDER BY revenue DESC;

Revenue Generated by Top 1% Restaurants
SELECT SUM(rating_count * cost)
FROM (
    SELECT rating_count, cost,
           ROW_NUMBER() OVER(ORDER BY rating_count * cost DESC) AS rank
    FROM restaurants
) t
WHERE t.rank <= (SELECT CEIL(COUNT(*) * 0.01) FROM restaurants);
